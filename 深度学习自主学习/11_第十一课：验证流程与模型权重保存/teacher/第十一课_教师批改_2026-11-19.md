### 完整参考代码

在第十一课目录运行，使用课程的 `practice_fixture.pt`。可整段用于参考副本。局部对照时，将对照选优的三项初值放在候选循环前，把第二套比较放在原正式比较之后，仍在候选循环内。两个选择结果在循环结束后报告。将原第 80 行起的末尾三行替换为参考中从 `best_weights` 开始的部分，依赖上文已定义的 `SmallClassifier`、`best_name`、`data`、`model` 和 `loss_fn`。代码会在工作目录写入 `practice_best.pt`。

```python
import torch  # 导入张量、权重文件读写和比较功能。
from torch import nn  # 导入模型层和损失函数。
from torch.utils.data import DataLoader, TensorDataset  # 导入验证集的逐批读取工具。


class SmallClassifier(nn.Module):  # 按候选权重所要求的结构建模。
    def __init__(self):  # 定义网络中的层。
        super().__init__()  # 初始化 PyTorch 模型基类。
        self.hidden = nn.Linear(2, 4)  # 两列输入特征映射为四个隐藏值。
        self.activation = nn.ReLU()  # 隐藏层使用 ReLU 激活。
        self.output = nn.Linear(4, 2)  # 输出两个类别的分数。

    def forward(self, inputs):  # 定义模型的一次前向计算。
        hidden = self.hidden(inputs)  # 计算隐藏层输出。
        activated = self.activation(hidden)  # 应用激活函数。
        return self.output(activated)  # 返回类别原始分数 logits。


# 在第十一课目录中运行，使用课程提供的文件。
data = torch.load('practice_fixture.pt', weights_only=True, map_location='cpu')  # 将张量和状态字典读到 CPU。
val_dataset = TensorDataset(data['val_features'], data['val_targets'])  # 将验证特征与标签配对。
val_loader = DataLoader(val_dataset, batch_size=4, shuffle=False, drop_last=False)  # 固定顺序并保留尾批。
model = SmallClassifier()  # 创建依次评估候选的模型。
loss_fn = nn.CrossEntropyLoss()  # 每次返回该批的平均交叉熵。

best_name = None  # 记录正式规则选中的候选。
best_acc = float('-inf')  # 按准确率优先比较。
best_avg_loss = float('inf')  # 准确率并列时，记录最小的按样本平均损失。
batch_best_name = None  # 老师新增：单独记录按批次平均的对照赢家。
batch_best_acc = float('-inf')  # 对照规则保留准确率优先。
batch_best_loss = float('inf')  # 对照规则只替换损失的汇总口径。

for name in data['candidates']:  # 依题面顺序评估每一组权重。
    model.load_state_dict(data['candidates'][name], strict=True)  # 加载当前候选权重并检查键与形状。
    model.eval()  # 让模型进入评估模式。
    total_loss = 0  # 累计各样本的损失总量。
    total = 0  # 累计实际处理的样本数。
    correct = 0  # 累计预测正确的个数。
    batch_loss_sum = 0  # 累计各批均值之和。
    batch_count = 0  # 记录实际走过的批次数。

    with torch.no_grad():  # 验证前向无需记录梯度。
        for features, targets in val_loader:  # 每个候选都使用同一个验证集。
            logits = model(features)  # 计算当前批的原始类别分数。
            pred = logits.argmax(dim=1)  # 选取每条样本分数最高的类别。
            correct += (pred == targets).sum().item()  # 统计预测与标签相同的个数。
            size = len(targets)  # 用实际批大小处理尾批。
            total += size  # 累加样本数。
            batch_loss = loss_fn(logits, targets).item()  # 取当前批平均损失。
            total_loss += batch_loss * size  # 还原这个批次的损失总量。
            batch_loss_sum += batch_loss  # 为批次等权对照保留一份累计量。
            batch_count += 1  # 当前候选又完成一批验证。

    acc = correct / total  # 算出整体验证准确率。
    avg_loss = total_loss / total  # 按样本平均作为正式比较指标。
    avg_batch_loss = batch_loss_sum / batch_count  # 将各批视为等权计算对照值。
    print(name, '样本数', total, '正确数', correct, '准确率', acc, '按样本平均', avg_loss, '按批次平均', avg_batch_loss)  # 给出候选级汇总。

    if acc > best_acc or (acc == best_acc and avg_loss < best_avg_loss):  # 正式规则优先准确率，再比较按样本平均的损失。
        best_name = name  # 记录暂时领先的候选。
        best_acc = acc  # 与名称同步保存准确率。
        best_avg_loss = avg_loss  # 与名称同步保存按样本平均的损失。

    if acc > batch_best_acc or (acc == batch_best_acc and avg_batch_loss < batch_best_loss):  # 老师新增：只改变损失的比较口径。
        batch_best_name = name  # 独立保存对照规则的候选名称。
        batch_best_acc = acc  # 更新该候选的准确率。
        batch_best_loss = avg_batch_loss  # 更新该候选的批次平均损失。

print('按样本平均选中', best_name)  # 输出正式选优结果。
print('按批次平均选中', batch_best_name)  # 输出对照结果。

best_weights = data['candidates'][best_name]  # 取正式赢家的状态字典。
torch.save(best_weights, 'practice_best.pt')  # 老师修改：将已有的状态字典写入文件，便于按课程方式读取。
model.load_state_dict(best_weights, strict=True)  # 重新将评估用模型切换到正式赢家。
model.eval()  # 确保原模型输出采用评估模式。
with torch.no_grad():  # 为重载保留同一输入下的全部类别分数。
    before_logits = model(data['val_features'])  # 获得赢家在整份验证集上的输出。

restored = SmallClassifier()  # 老师新增：新建独立模型以核对文件能否恢复模型状态。
loaded_weights = torch.load('practice_best.pt', weights_only=True, map_location='cpu')  # 读回刚保存的状态字典。
restored.load_state_dict(loaded_weights, strict=True)  # 严格加载文件内容。
restored.eval()  # 显式将新模型设为评估模式。
with torch.no_grad():  # 比较同一份验证数据上的完整输出。
    after_logits = restored(data['val_features'])  # 计算重载后的全部验证分数。
same_logits = torch.allclose(before_logits, after_logits, atol=1e-6, rtol=0)  # 用课程约定的容差比较每个元素。
print('重载后全部验证分数一致', same_logits)  # 报告重载核验结果。
assert same_logits  # 确认重载核对通过后进入最终测试。

with torch.no_grad():  # 老师新增：完成选优和重载后才读取测试指标。
    test_logits = restored(data['test_features'])  # 使用重载模型前向计算测试分数。
    test_targets = data['test_targets']  # 取测试集的真实类别。
    test_predictions = test_logits.argmax(dim=1)  # 从每行分数取出预测类别。
    test_loss = loss_fn(test_logits, test_targets).item()  # 求全部测试样本的平均损失。
    test_correct = (test_predictions == test_targets).sum().item()  # 统计测试预测正确的样本数。
    test_acc = test_correct / len(test_targets)  # 将正确数换算为准确率。
print('测试预测类别', test_predictions.tolist())  # 报告每条测试样本的预测。
print('测试平均损失', test_loss)  # 报告测试集平均损失。
print('测试准确率', test_acc)  # 报告测试集准确率。
```

### 文字参考

1. **预测。** 两种平均的排序未必相同。批次大小不等时，两种算法给各批分配的权重不同。准确率并列只说明答对数量相同。
2. **这次为何改选。** 首批四条，尾批一条。按批次平均时两批各占一半，尾批影响增大。B 在首批损失比 A 低，在尾批损失比 A 高。实跑按样本平均选中 B，按批次平均选中 A。
3. **交叉熵为何能区分。** 准确率只统计最高分对应的类别是否命中。交叉熵还取决于真实类别获得的概率，即使两模型选中同一类别，概率也可以不同。
4. **测试集的使用时机。** 先用验证集完成选优，最后用测试集评价选定的权重。如果根据测试结果挑选候选，测试数据也参与了选择，报告的表现可能偏乐观。
5. **两种验证设置。** `eval()` 切换模型层的运行模式，例如让 Dropout 停止随机丢弃。`torch.no_grad()` 关闭所在代码块的梯度记录。
