## 完整修正版本与位置

课程文档【我的练习解答】已同步下面的现有 `test.py`；源码无需再改。创建目录位于路径定义后，输出命名在 `make_output_path` 内，哈希计算位于保存后的打印循环内。以下逐行注释版用于对照学习，执行逻辑与现有源码一致。

```python
import os  # 定义或执行图片处理步骤
import hashlib  # 定义或执行图片处理步骤
from torchvision import transforms as T  # 定义或执行图片处理步骤
import torch  # 定义或执行图片处理步骤
from PIL import Image  # 定义或执行图片处理步骤

BASE_DIR = os.path.dirname(os.path.abspath(__file__))  # 定义或执行图片处理步骤
INPUT_DIR = os.path.join(BASE_DIR, 'input')  # 定义或执行图片处理步骤
OUTPUT_DIR = os.path.join(BASE_DIR, 'output')  # 定义或执行图片处理步骤

IMAGE_SUFFIXES = {'.jpg', '.jpeg', '.png', '.webp'}  # 定义或执行图片处理步骤
BASE_SEED = 73  # 定义或执行图片处理步骤
os.makedirs(OUTPUT_DIR, exist_ok=True)  # 定义或执行图片处理步骤
transform = T.Compose([  # 定义或执行图片处理步骤
	T.Resize((160, 160)),  # 定义或执行图片处理步骤
	T.RandomHorizontalFlip(p=0.5),  # 定义或执行图片处理步骤
	T.RandomRotation(degrees=20, fill=(0, 0, 0)),  # 定义或执行图片处理步骤
	T.ColorJitter(brightness=0.3),  # 定义或执行图片处理步骤
	T.GaussianBlur(kernel_size=3, sigma=0.5)  # 定义或执行图片处理步骤
])  # 结束当前参数或变换列表


def get_image_files(input_dir):  # 定义或执行图片处理步骤
	image_paths = []  # 定义或执行图片处理步骤
	for file_name in sorted(os.listdir(input_dir)):  # 定义或执行图片处理步骤
		file_path = os.path.join(input_dir, file_name)  # 定义或执行图片处理步骤
		suffix = os.path.splitext(file_name)[1].lower()  # 定义或执行图片处理步骤
		if os.path.isfile(file_path) and suffix in IMAGE_SUFFIXES:  # 定义或执行图片处理步骤
			image_paths.append(file_path)  # 定义或执行图片处理步骤
		else:  # 定义或执行图片处理步骤
			print('skip', file_name)  # 定义或执行图片处理步骤
	return image_paths  # 定义或执行图片处理步骤


def make_output_path(file_path, output_dir):  # 定义函数：创建不冲突的输出文件路径
	base_name = os.path.splitext(os.path.basename(file_path))[0]  # 函数⑤：取输入图片的主文件名
	output_name = base_name + '_aug.png'  # 给增强结果添加后缀
	output_path = os.path.join(output_dir, output_name)  # 函数②：拼接输出路径
	number = 1  # 设置冲突文件的起始编号
	while os.path.exists(output_path):  # 函数⑨：检查目标文件是否已经存在
		output_name = base_name + '_aug_' + str(number) + '.png'  # 生成带编号的新文件名
		output_path = os.path.join(output_dir, output_name)  # 函数②：更新目标路径
		number = number + 1  # 编号加一，继续检查下一个名称
	return output_path  # 返回不会覆盖已有文件的路径


def enhance_one_image(file_path, output_dir, image_index):  # 定义或执行图片处理步骤
	torch.manual_seed(BASE_SEED + image_index)  # 定义或执行图片处理步骤
	with Image.open(file_path) as source_image:  # 定义或执行图片处理步骤
		image = source_image.convert('RGB')  # 定义或执行图片处理步骤
	enhanced_image = transform(image)  # 定义或执行图片处理步骤
	output_path = make_output_path(file_path, output_dir)  # 定义或执行图片处理步骤
	enhanced_image.save(output_path)  # 定义或执行图片处理步骤
	return output_path, enhanced_image.size, enhanced_image.mode  # 定义或执行图片处理步骤


image_paths = get_image_files(INPUT_DIR)  # 收集输入目录中的图片
print('筛选后图片数量：', len(image_paths))  # 输出筛选数量
for image_index in range(len(image_paths)):  # 定义或执行图片处理步骤
	output_path, output_size, output_mode = enhance_one_image(  # 定义或执行图片处理步骤
		image_paths[image_index], OUTPUT_DIR, image_index  # 定义或执行图片处理步骤
	)  # 结束当前参数或变换列表
	print('已保存：', os.path.basename(output_path))  # 输出保存后的文件名
	print('输出尺寸：', output_size)  # 输出图像实际尺寸
	print('输出模式：', output_mode)  # 输出图像颜色模式
	with open(output_path, 'rb') as image_file:  # 以二进制只读方式打开生成图片
		image_hash = hashlib.sha256(image_file.read()).hexdigest()  # 函数⑨：计算图片SHA-256
	print('SHA-256：', image_hash)  # 输出图片的字节指纹
```
