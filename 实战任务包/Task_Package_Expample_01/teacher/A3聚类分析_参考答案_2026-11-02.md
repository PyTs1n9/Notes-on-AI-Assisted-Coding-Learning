- 肘部法则：inertia 逐 K 降幅 = [396.3, 202.2, 175.6, 116.7, 101.4, 70.2]（依次对应 K=3…8），降幅到 K=4 还有 202.2，**K=5 起跌破 200 并一路收窄**（175.6→116.7→101.4→70.2）→ 拐点落在 K=4；
- 轮廓系数：全局最大在 K=2（0.301661），之后一路走低到 K=6（0.225555），K=7、K=8 只微升到 0.229 上下；**K=4（0.264953）高于 K=5~8 的全部取值**（0.237236／0.225555／0.229247／0.229450），也就是 K 再往上分，簇只会更碎；

import pandas as pd

from sklearn.preprocessing import StandardScaler

# ---------- 一、K=2~8 三指标（和你 kmeans.py 里的循环同一套写法）----------
Ks = list(range(2, 9))
inertia_list = []
sil_list = []
dbi_list = []

for k in Ks:
    km = KMeans(n_clusters=k, random_state=42, n_init=10)
    km.fit(X_scaled)
    inertia_list.append(km.inertia_)                                  # Inertia
    sil_list.append(silhouette_score(X_scaled, km.labels_))           # 轮廓系数
    dbi_list.append(davies_bouldin_score(X_scaled, km.labels_))       # DBI

table = pd.DataFrame({'K': Ks, 'inertia': inertia_list, 'sil': sil_list, 'dbi': dbi_list})
print('--- K=2~8 三指标 ---')
print(table.round(6).to_string(index=False))    # to_string(index=False) = 不打印左边的行号

best_sil_k = Ks[0]
best_sil = sil_list[0]

best_dbi = dbi_list[0]

    if sil_list[j] > best_sil:

        best_sil_k = Ks[j]
    if dbi_list[j] < best_dbi:
        best_dbi = dbi_list[j]
        best_dbi_k = Ks[j]

drop_list = []                                        # inertia 逐 K 降幅
for j in range(1, len(inertia_list)):

print('轮廓系数最大的 K =', best_sil_k, ' 值 =', round(best_sil, 6))
print('DBI 最小的 K =', best_dbi_k, ' 值 =', round(best_dbi, 6))

plt.xlabel('temp')

cluster_list = []
temp_list = []
hum_list = []
wind_list = []

count_list = []

for c in range(K):                                    # c = 0,1,2,3 逐个簇算
    mask = (labels == c)                              # 731 个 True/False，标出哪些天属于簇 c
    cluster_list.append(c)
    temp_list.append(round(df.loc[mask, 'temp'].mean(), 6))          # 该簇 temp 均值
    hum_list.append(round(df.loc[mask, 'hum'].mean(), 6))            # 该簇 hum 均值

cluster_summary = pd.DataFrame({

cluster_summary.to_csv('../data_outputs/cluster_summary.csv', index=False, encoding='utf-8-sig')
print('--- cluster_summary.csv ---')
print(cluster_summary.to_string(index=False))

# ---------- 六、A3-5 命名依据：先算基线，再逐簇对比 ----------

for c in range(K):
    mask = (labels == c)
